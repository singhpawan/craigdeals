CREATE NETWORK POLICY TEMP_ALLOW_ALL
  ALLOWED_IP_LIST = ('0.0.0.0/0'); -- Allows all IP addresses

ALTER USER YOUR_SNOWFLAKE_USER SET NETWORK_POLICY = TEMP_ALLOW_ALL;

-- Remove the temporary policy from the user
ALTER USER YOUR_SNOWFLAKE_USER UNSET NETWORK_POLICY;

-- or, assign the original network policy.
ALTER USER YOUR_SNOWFLAKE_USER SET NETWORK_POLICY = YOUR_ORIGINAL_NETWORK_POLICY;

-- Drop the temporary policy
DROP NETWORK POLICY TEMP_ALLOW_ALL;
