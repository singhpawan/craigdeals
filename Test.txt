SELECT
    EVENT_TIMESTAMP,
    USER_NAME,
    CLIENT_IP,
    ERROR_CODE,
    ERROR_MESSAGE,
    AUTHENTICATION_METHOD,
    ROLE_NAME,
    WAREHOUSE_NAME,
    FIRST_AUTHENTICATION_FACTOR,
    SECOND_AUTHENTICATION_FACTOR
FROM
    SNOWFLAKE.ACCOUNT_USAGE.AUTHENTICATION_HISTORY
WHERE
    USER_NAME = 'YOUR_SNOWFLAKE_USER' -- Replace with the actual username
    AND EVENT_TIMESTAMP >= DATEADD(hour, -24, CURRENT_TIMESTAMP()) -- Look back 24 hours (adjust as needed)
ORDER BY
    EVENT_TIMESTAMP DESC;
