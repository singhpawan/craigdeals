Here's a clean query for that:
sql-- Set your parameters
SET root_task_name = 'my_db.my_schema.MY_ROOT_TASK';
SET run_date = '2026-03-15';

-- Get root task ID
SET root_task_id = (
    SELECT "id" FROM TABLE(RESULT_SCAN(LAST_QUERY_ID()))
    -- Or hardcode it if you already know it
);

-- Full DAG run status for a given date
SELECT
    h.NAME                          AS task_name,
    h.STATE,
    h.CONDITION_TEXT,
    h.GRAPH_RUN_GROUP_ID,
    h.SCHEDULED_TIME,
    h.QUERY_START_TIME,
    h.COMPLETED_TIME,
    DATEDIFF('second', h.QUERY_START_TIME, h.COMPLETED_TIME) AS duration_secs,
    h.ERROR_CODE,
    h.ERROR_MESSAGE,
    h.QUERY_ID,
    h.ATTEMPT_NUMBER
FROM TABLE(INFORMATION_SCHEMA.TASK_HISTORY(
    ROOT_TASK_ID => '<your_root_task_id>',
    SCHEDULED_TIME_RANGE_START => $run_date::TIMESTAMP_LTZ,
    SCHEDULED_TIME_RANGE_END   => DATEADD('day', 1, $run_date::TIMESTAMP_LTZ)
)) h
ORDER BY h.GRAPH_RUN_GROUP_ID, h.SCHEDULED_TIME;
If you want a summary per DAG run for that day:
sqlSELECT
    GRAPH_RUN_GROUP_ID,
    MIN(SCHEDULED_TIME)                                          AS dag_started,
    MAX(COMPLETED_TIME)                                          AS dag_ended,
    DATEDIFF('minute', MIN(SCHEDULED_TIME), MAX(COMPLETED_TIME)) AS total_duration_min,
    COUNT(*)                                                     AS total_tasks,
    SUM(CASE WHEN STATE = 'SUCCEEDED' THEN 1 ELSE 0 END)        AS succeeded,
    SUM(CASE WHEN STATE = 'FAILED'    THEN 1 ELSE 0 END)        AS failed,
    SUM(CASE WHEN STATE = 'SKIPPED'   THEN 1 ELSE 0 END)        AS skipped,
    SUM(CASE WHEN STATE = 'CANCELLED' THEN 1 ELSE 0 END)        AS cancelled,
    LISTAGG(
        CASE WHEN STATE = 'FAILED' THEN NAME END, ', '
    )                                                            AS failed_task_names,
    LISTAGG(
        CASE WHEN STATE = 'FAILED' THEN ERROR_MESSAGE END, ' | '
    )                                                            AS failure_reasons
FROM TABLE(INFORMATION_SCHEMA.TASK_HISTORY(
    ROOT_TASK_ID => '<your_root_task_id>',
    SCHEDULED_TIME_RANGE_START => '2026-03-15'::TIMESTAMP_LTZ,
    SCHEDULED_TIME_RANGE_END   => '2026-03-16'::TIMESTAMP_LTZ
))
GROUP BY GRAPH_RUN_GROUP_ID
ORDER BY dag_started;
If the task runs multiple times a day and you want to compare all runs side by side:
sqlSELECT
    GRAPH_RUN_GROUP_ID,
    NAME                 AS task_name,
    STATE,
    CASE
        WHEN STATE = 'SUCCEEDED' THEN 'Pass'
        WHEN STATE = 'FAILED'    THEN 'FAIL - ' || NVL(ERROR_MESSAGE, 'unknown')
        WHEN STATE = 'SKIPPED'   THEN 'Skipped (condition not met)'
        WHEN STATE = 'CANCELLED' THEN 'Cancelled'
        ELSE STATE
    END                  AS status_detail,
    SCHEDULED_TIME,
    DATEDIFF('second', QUERY_START_TIME, COMPLETED_TIME) AS duration_secs,
    QUERY_ID
FROM TABLE(INFORMATION_SCHEMA.TASK_HISTORY(
    ROOT_TASK_ID => '<your_root_task_id>',
    SCHEDULED_TIME_RANGE_START => '2026-03-15'::TIMESTAMP_LTZ,
    SCHEDULED_TIME_RANGE_END   => '2026-03-16'::TIMESTAMP_LTZ
))
ORDER BY GRAPH_RUN_GROUP_ID, SCHEDULED_TIME;

Step 1: Get full query details
sql-- Core query details
SELECT QUERY_ID, QUERY_TEXT, QUERY_TYPE,
       DATABASE_NAME, SCHEMA_NAME, WAREHOUSE_NAME, WAREHOUSE_SIZE,
       EXECUTION_STATUS, ERROR_CODE, ERROR_MESSAGE,
       START_TIME, END_TIME,
       DATEDIFF('second', START_TIME, END_TIME) AS duration_secs,
       TOTAL_ELAPSED_TIME,         -- in milliseconds
       BYTES_SCANNED,
       ROWS_PRODUCED,
       ROWS_INSERTED, ROWS_UPDATED, ROWS_DELETED,
       PARTITIONS_SCANNED, PARTITIONS_TOTAL,
       BYTES_WRITTEN_TO_RESULT,
       COMPILATION_TIME,
       EXECUTION_TIME,
       QUEUED_PROVISIONING_TIME,   -- time waiting for warehouse
       QUEUED_OVERLOAD_TIME,       -- time queued due to overload
       CREDITS_USED_CLOUD_SERVICES
FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
WHERE QUERY_ID = '<your_query_id>';
Step 2: Link the query ID back to its task run
sql-- Find the task execution that produced this query
SELECT NAME AS task_name,
       DATABASE_NAME, SCHEMA_NAME,
       STATE,
       SCHEDULED_TIME, QUERY_START_TIME, COMPLETED_TIME,
       DATEDIFF('second', QUERY_START_TIME, COMPLETED_TIME) AS actual_run_secs,
       CONDITION_TEXT,        -- The WHEN condition
       ERROR_CODE, ERROR_MESSAGE,
       QUERY_ID,
       GRAPH_RUN_GROUP_ID,
       ROOT_TASK_ID,
       ATTEMPT_NUMBER,
       CONFIG                 -- Task config like warehouse, schedule
FROM TABLE(INFORMATION_SCHEMA.TASK_HISTORY(
    SCHEDULED_TIME_RANGE_START => DATEADD('day', -7, CURRENT_TIMESTAMP())
))
WHERE QUERY_ID = '<your_query_id>';
The CONDITION_TEXT column here tells you exactly what WHEN condition was evaluated before the task ran.
Step 3: Check if the condition query itself was logged
When a task has a WHEN condition, Snowflake actually runs a separate query to evaluate it before the main query. You can find it:
sql-- Look for the condition evaluation query
-- It typically runs just before the main query, in the same session
SELECT QUERY_ID, QUERY_TEXT, QUERY_TYPE,
       EXECUTION_STATUS, START_TIME, END_TIME,
       ROWS_PRODUCED
FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
WHERE SESSION_ID = (
    SELECT SESSION_ID
    FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
    WHERE QUERY_ID = '<your_query_id>'
)
AND START_TIME BETWEEN
    DATEADD('minute', -5, (
        SELECT START_TIME FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
        WHERE QUERY_ID = '<your_query_id>'
    ))
    AND (
        SELECT START_TIME FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
        WHERE QUERY_ID = '<your_query_id>'
    )
ORDER BY START_TIME;
Step 4: Get the query profile (execution plan details)
sql-- Execution plan breakdown
SELECT *
FROM TABLE(GET_QUERY_OPERATOR_STATS('<your_query_id>'));
This shows you every operator in the execution plan with stats like rows processed, bytes spilled, partitions scanned per operator — very useful for understanding why it was fast (e.g., pruned all partitions, empty source table).
Step 5: Full picture — task run + all child tasks + condition
sql-- First get the GRAPH_RUN_GROUP_ID and ROOT_TASK_ID
SET graph_id = (
    SELECT GRAPH_RUN_GROUP_ID
    FROM TABLE(INFORMATION_SCHEMA.TASK_HISTORY(
        SCHEDULED_TIME_RANGE_START => DATEADD('day', -7, CURRENT_TIMESTAMP())
    ))
    WHERE QUERY_ID = '<your_query_id>'
);

-- Now get the entire DAG run
SELECT NAME,
       STATE,
       CONDITION_TEXT,
       SCHEDULED_TIME,
       QUERY_START_TIME,
       COMPLETED_TIME,
       DATEDIFF('second', QUERY_START_TIME, COMPLETED_TIME) AS run_secs,
       ERROR_MESSAGE,
       QUERY_ID,
       ATTEMPT_NUMBER
FROM TABLE(INFORMATION_SCHEMA.TASK_HISTORY(
    ROOT_TASK_ID => '<root_task_id>',
    SCHEDULED_TIME_RANGE_START => DATEADD('day', -7, CURRENT_TIMESTAMP())
))
WHERE GRAPH_RUN_GROUP_ID = $graph_id
ORDER BY SCHEDULED_TIME;
Step 6: Check if the task was manually triggered vs scheduled
sql-- The TRIGGERED_BY column (available in newer Snowflake versions)
SELECT NAME, STATE, SCHEDULED_TIME,
       CONDITION_TEXT,
       QUERY_ID
FROM SNOWFLAKE.ACCOUNT_USAGE.TASK_HISTORY
WHERE QUERY_ID = '<your_query_id>'
   OR GRAPH_RUN_GROUP_ID = (
       SELECT GRAPH_RUN_GROUP_ID
       FROM SNOWFLAKE.ACCOUNT_USAGE.TASK_HISTORY
       WHERE QUERY_ID = '<your_query_id>'
   )
ORDER BY SCHEDULED_TIME;

This combination gives you complete observability — the condition that was evaluated, whether it passed, what the query actually did, and what happened downstream.
